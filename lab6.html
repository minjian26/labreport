import numpy as np
import matplotlib.pyplot as plt

def generate_grid(size):
    x = np.linspace(-1, , size)  
    y = np.linspace(-1, 1, size)
    grid = np.meshgrid(x, y) 
    return grid
		
def plot_grid(grid):
    x_grid, y_grid = grid
    plt.figure(figsize=(10, 10))
    plt.scatter(x_grid, y_grid, s=1, color='blue') 
    plt.title(f'2D Grid of {len(x_grid)}x{len(y_grid)} points')
    plt.show()

size = 20 

grid = generate_grid(size)
plot_grid(grid)

Next, based on the above code, it is imported an iterative equation to create a 3D grid of points. It is introduced a complex plane and set the iterative equation. It can be set the number of iterations. I try to set the number of iterations is 3. The grid of points are scattered. The followings are the comparation. 


def iterative_transform(grid, iterations):
    x_grid, y_grid = grid
    z = x_grid + 1j * y_grid 
    for i in range(iterations):
        z = z ** 2 - 0.5 
    return np.real(z), np.imag(z)

# Set the number of iterations
iterations = 1

# Apply the iterative transformation
fractal_grid = iterative_transform(grid, iterations)

# Plot the fractal grid
plot_grid(fractal_grid)
